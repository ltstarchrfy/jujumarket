   148→interface Particle {
   149→  x: number;
   150→  y: number;
   151→  size: number;
   152→  speedY: number;
   153→  speedX: number;
   154→  rotation: number;
   155→  rotSpeed: number;
   156→  swayAmp: number;

   158→  swayOffset: number;

   164→// ─── Particle Canvas ─────────────────────────────────────────────────────────
   165→
   166→function ParticleCanvas() {
   167→  const canvasRef = useRef<HTMLCanvasElement>(null);
   168→  const particlesRef = useRef<Particle[]>([]);
   169→  const timeRef = useRef(0);
   170→  const animRef = useRef<number>(0);
   171→
   172→  useEffect(() => {
   173→    const canvas = canvasRef.current;
   174→    if (!canvas) return;
   175→    const ctx = canvas.getContext("2d");
   176→    if (!ctx) return;
   177→
   178→    let w = 0, h = 0;
   179→
   180→    function resize() {
   181→      if (!canvas) return;
   182→      w = canvas.width = window.innerWidth;
   183→      h = canvas.height = window.innerHeight;
   184→    }
   185→    resize();
   186→    window.addEventListener("resize", resize);
   187→
   188→    const isMobile = window.innerWidth < 768;
   189→    const count = isMobile ? 25 : 55;
   190→
   191→    function createParticle(init: boolean): Particle {
   192→      return {
   193→        x: Math.random() * w,
   194→        y: init ? Math.random() * h : -20,
   195→        size: Math.random() * 5 + 2,
   196→        speedY: Math.random() * 0.8 + 0.3,
   197→        speedX: Math.random() * 0.4 - 0.2,
   198→        rotation: Math.random() * Math.PI * 2,
   199→        rotSpeed: (Math.random() - 0.5) * 0.02,
   200→        swayAmp: Math.random() * 0.8 + 0.3,
   201→        swaySpeed: Math.random() * 0.015 + 0.005,
   202→        swayOffset: Math.random() * Math.PI * 2,
   203→        alpha: Math.random() * 0.25 + 0.05,
   204→        type: Math.floor(Math.random() * 3),
   205→        color:
   206→          Math.random() < 0.5
   207→            ? "37,99,235"
   208→            : Math.random() < 0.8
   209→              ? "96,165,250"
   210→              : "34,211,238",
   211→      };
   212→    }
   213→
   214→    const leaves: Particle[] = [];
   215→    for (let i = 0; i < count; i++) leaves.push(createParticle(true));
   216→    particlesRef.current = leaves;
   217→
   218→    function drawParticle(p: Particle, t: number) {
   219→      if (!ctx) return;
   220→      ctx.save();
   221→      ctx.translate(p.x, p.y);
   222→      ctx.rotate(p.rotation);
   223→      ctx.globalAlpha = p.alpha;
   224→
   225→      if (p.type === 0) {
   226→        ctx.beginPath();
   227→        ctx.moveTo(0, -p.size);
   228→        ctx.bezierCurveTo(p.size * 0.8, -p.size * 0.5, p.size * 0.8, p.size * 0.5, 0, p.size);
   229→        ctx.bezierCurveTo(-p.size * 0.8, p.size * 0.5, -p.size * 0.8, -p.size * 0.5, 0, -p.size);
   230→        ctx.fillStyle = `rgba(${p.color},1)`;
   231→        ctx.fill();
   232→      } else if (p.type === 1) {
   233→        ctx.beginPath();
   234→        ctx.arc(0, 0, p.size * 0.5, 0, Math.PI * 2);
   235→        ctx.fillStyle = `rgba(${p.color},1)`;
   236→        ctx.fill();
   237→      } else {
   238→        const s = p.size * 0.6;
   239→        ctx.strokeStyle = `rgba(${p.color},1)`;
   240→        ctx.lineWidth = 0.8;
   241→        ctx.beginPath(); ctx.moveTo(-s, 0); ctx.lineTo(s, 0); ctx.stroke();
   242→        ctx.beginPath(); ctx.moveTo(0, -s); ctx.lineTo(0, s); ctx.stroke();
   243→        ctx.beginPath(); ctx.moveTo(-s * 0.6, -s * 0.6); ctx.lineTo(s * 0.6, s * 0.6); ctx.stroke();
   244→        ctx.beginPath(); ctx.moveTo(s * 0.6, -s * 0.6); ctx.lineTo(-s * 0.6, s * 0.6); ctx.stroke();
   245→      }
   246→      ctx.globalAlpha = 1;
   247→      ctx.restore();
   248→    }
   249→
   250→    function loop() {
   251→      if (!ctx || !canvas) return;
   252→      ctx.clearRect(0, 0, w, h);
   253→      timeRef.current++;
   254→      const t = timeRef.current;
   255→      for (const p of particlesRef.current) {
   256→        p.y += p.speedY;
   257→        p.x += p.speedX + Math.sin(t * p.swaySpeed + p.swayOffset) * p.swayAmp * 0.3;
   258→        p.rotation += p.rotSpeed;
   259→        if (p.y > h + 20) Object.assign(p, createParticle(false));
   260→        if (p.x < -20) p.x = w + 20;
   261→        if (p.x > w + 20) p.x = -20;
   262→        drawParticle(p, t);
   263→      }
   264→      animRef.current = requestAnimationFrame(loop);
   265→    }
   266→    loop();
   267→
   268→    return () => {
   269→      window.removeEventListener("resize", resize);
   270→      cancelAnimationFrame(animRef.current);
   271→    };
   272→  }, []);
   273→
   274→  return <canvas ref={canvasRef} className="fixed inset-0 z-[1] pointer-events-none" aria-hidden="true" />;
   275→}
