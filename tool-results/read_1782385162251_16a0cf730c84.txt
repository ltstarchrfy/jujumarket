   219→// ─── QR Code Placeholder ──────────────────────────────────────────────────────
   220→
   221→function QRPlaceholder() {
   222→  const cells = Array.from({ length: 121 }, (_, i) => {
   223→    const row = Math.floor(i / 11);
   224→    const col = i % 11;
   225→    const isCorner = (row < 3 && col < 3) || (row < 3 && col > 7) || (row > 7 && col < 3);
   226→    const isFilled = isCorner || Math.random() > 0.5;
   227→    return isFilled;
   228→  });
   229→
   230→  return (
   231→    <div className="inline-grid grid-cols-11 gap-[2px] p-4 bg-white rounded-xl">
   232→      {cells.map((filled, i) => (
   233→        <div
   234→          key={i}
   235→          className="w-4 h-4 sm:w-5 sm:h-5 rounded-[2px]"
   236→          style={{ background: filled ? '#1a1a2e' : '#f0f0f0' }}
   237→        />
   238→      ))}
   239→    </div>
   240→  );
   241→}

   253→// ─── Countdown Timer ──────────────────────────────────────────────────────────
   254→
   255→function CountdownTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
   256→  const [timeLeft, setTimeLeft] = useState(seconds);
   257→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   258→
   259→  useEffect(() => {
   260→    intervalRef.current = setInterval(() => {
   261→      setTimeLeft((prev) => {
   262→        if (prev <= 1) {
   263→          if (intervalRef.current) clearInterval(intervalRef.current);
   264→          onExpire();
   265→          return 0;
   266→        }
   267→        return prev - 1;
   268→      });
   269→    }, 1000);
   270→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   271→  }, [onExpire]);
   272→
   273→  const mins = Math.floor(timeLeft / 60);
   274→  const secs = timeLeft % 60;
   275→
   276→  return (
   277→    <div className="text-center">
   278→      <div className="text-3xl font-bold font-mono text-amber-400">
   279→        {String(mins).padStart(2, '0')}:{String(secs).padStart(2, '0')}
   280→      </div>
   281→      <p className="text-xs text-gray-400 mt-1">Sisa waktu pembayaran</p>
   282→    </div>
   283→  );
   284→}
