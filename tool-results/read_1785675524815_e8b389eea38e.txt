   166→// ─── Particle Canvas ─────────────────────────────────────────────────────────
   167→
   168→function ParticleCanvas() {
   169→  const canvasRef = useRef<HTMLCanvasElement>(null);
   170→  const particlesRef = useRef<Particle[]>([]);
   171→  const timeRef = useRef(0);
   172→  const animRef = useRef<number>(0);
   173→
   174→  useEffect(() => {
   175→    const canvas = canvasRef.current;
   176→    if (!canvas) return;
   177→    const ctx = canvas.getContext("2d", { alpha: true });
   178→    if (!ctx) return;
   179→
   180→    let w = 0, h = 0;
   181→    let running = true;
   182→    let lastTime = 0;
   183→    const TARGET_FPS = 30;
   184→    const FRAME_INTERVAL = 1000 / TARGET_FPS;
   185→
   186→    function resize() {
   187→      if (!canvas) return;
   188→      const dpr = Math.min(window.devicePixelRatio || 1, 2);
   189→      w = window.innerWidth;
   190→      h = window.innerHeight;
   191→      canvas.width = w * dpr;
   192→      canvas.height = h * dpr;
   193→      canvas.style.width = w + "px";
   194→      canvas.style.height = h + "px";
   195→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   196→    }
   197→    resize();
   198→    window.addEventListener("resize", resize);
   199→
   200→    const isMobile = window.innerWidth < 768;
   201→    const count = isMobile ? 12 : 22;
   202→
   203→    function createParticle(init: boolean): Particle {
   204→      return {
   205→        x: Math.random() * w,
   206→        y: init ? Math.random() * h : -20,
   207→        size: Math.random() * 4 + 1.5,
   208→        speedY: Math.random() * 0.5 + 0.15,
   209→        speedX: Math.random() * 0.25 - 0.12,
   210→        rotation: Math.random() * Math.PI * 2,
   211→        rotSpeed: (Math.random() - 0.5) * 0.012,
   212→        swayAmp: Math.random() * 0.5 + 0.2,
   213→        swaySpeed: Math.random() * 0.01 + 0.003,
   214→        swayOffset: Math.random() * Math.PI * 2,
   215→        alpha: Math.random() * 0.18 + 0.04,
   216→        type: Math.floor(Math.random() * 3),
   217→        color:
   218→          Math.random() < 0.5
   219→            ? "37,99,235"
   220→            : Math.random() < 0.8
   221→              ? "96,165,250"
   222→              : "34,211,238",
   223→      };
   224→    }
   225→
   226→    const leaves: Particle[] = [];
   227→    for (let i = 0; i < count; i++) leaves.push(createParticle(true));
   228→    particlesRef.current = leaves;
   229→
   230→    function drawParticle(p: Particle) {
   231→      if (!ctx) return;
   232→      ctx.globalAlpha = p.alpha;
   233→
   234→      if (p.type === 0) {
   235→        ctx.save();
   236→        ctx.translate(p.x, p.y);
   237→        ctx.rotate(p.rotation);
   238→        ctx.beginPath();
   239→        ctx.moveTo(0, -p.size);
   240→        ctx.bezierCurveTo(p.size * 0.8, -p.size * 0.5, p.size * 0.8, p.size * 0.5, 0, p.size);
   241→        ctx.bezierCurveTo(-p.size * 0.8, p.size * 0.5, -p.size * 0.8, -p.size * 0.5, 0, -p.size);
   242→        ctx.fillStyle = `rgba(${p.color},1)`;
   243→        ctx.fill();
   244→        ctx.restore();
   245→      } else if (p.type === 1) {
   246→        ctx.beginPath();
   247→        ctx.arc(p.x, p.y, p.size * 0.5, 0, Math.PI * 2);
   248→        ctx.fillStyle = `rgba(${p.color},1)`;
   249→        ctx.fill();
   250→      } else {
   251→        const s = p.size * 0.6;
   252→        const cx = p.x, cy = p.y;
   253→        ctx.save();
   254→        ctx.translate(cx, cy);
   255→        ctx.rotate(p.rotation);
   256→        ctx.strokeStyle = `rgba(${p.color},1)`;
   257→        ctx.lineWidth = 0.8;
   258→        ctx.beginPath(); ctx.moveTo(-s, 0); ctx.lineTo(s, 0); ctx.stroke();
   259→        ctx.beginPath(); ctx.moveTo(0, -s); ctx.lineTo(0, s); ctx.stroke();
   260→        ctx.restore();
   261→      }
   262→      ctx.globalAlpha = 1;
   263→    }
   264→
   265→    function loop(timestamp: number) {
   266→      if (!running) return;
   267→      animRef.current = requestAnimationFrame(loop);
   268→      
   269→      const delta = timestamp - lastTime;
   270→      if (delta < FRAME_INTERVAL) return;
   271→      lastTime = timestamp - (delta % FRAME_INTERVAL);
   272→      
   273→      if (!ctx || !canvas) return;
   274→      ctx.clearRect(0, 0, w, h);
   275→      timeRef.current++;
   276→      const t = timeRef.current;
   277→      for (const p of particlesRef.current) {
   278→        p.y += p.speedY;
   279→        p.x += p.speedX + Math.sin(t * p.swaySpeed + p.swayOffset) * p.swayAmp * 0.3;
   280→        p.rotation += p.rotSpeed;
   281→        if (p.y > h + 20) Object.assign(p, createParticle(false));
   282→        if (p.x < -20) p.x = w + 20;
   283→        if (p.x > w + 20) p.x = -20;
   284→        drawParticle(p);
   285→      }
   286→    }
   287→    animRef.current = requestAnimationFrame(loop);
   288→
   289→    return () => {
   290→      running = false;
   291→      window.removeEventListener("resize", resize);
   292→      cancelAnimationFrame(animRef.current);
   293→    };
   294→  }, []);
   295→
   296→  return <canvas ref={canvasRef} className="fixed inset-0 z-[1] pointer-events-none" aria-hidden="true" style={{ willChange: "transform" }} />;
   297→}
