   146→interface Particle {
   147→  x: number;
   148→  y: number;
   149→  size: number;
   150→  speedY: number;
   151→  speedX: number;
   152→  rotation: number;
   153→  rotSpeed: number;
   154→  swayAmp: number;
   155→  swaySpeed: number;
   156→  swayOffset: number;
   157→  alpha: number;
   158→  type: number;
   159→  color: string;

   162→// ─── Particle Canvas ─────────────────────────────────────────────────────────
   163→
   164→function ParticleCanvas() {
   165→  const canvasRef = useRef<HTMLCanvasElement>(null);
   166→  const particlesRef = useRef<Particle[]>([]);
   167→  const timeRef = useRef(0);
   168→  const animRef = useRef<number>(0);
   169→
   170→  useEffect(() => {
   171→    const canvas = canvasRef.current;
   172→    if (!canvas) return;
   173→    const ctx = canvas.getContext("2d");
   174→    if (!ctx) return;
   175→
   176→    let w = 0, h = 0;
   177→
   178→    function resize() {
   179→      if (!canvas) return;
   180→      w = canvas.width = window.innerWidth;
   181→      h = canvas.height = window.innerHeight;
   182→    }
   183→    resize();
   184→    window.addEventListener("resize", resize);
   185→
   186→    const isMobile = window.innerWidth < 768;
   187→    const count = isMobile ? 25 : 55;
   188→
   189→    function createParticle(init: boolean): Particle {
   190→      return {
   191→        x: Math.random() * w,
   192→        y: init ? Math.random() * h : -20,
   193→        size: Math.random() * 5 + 2,
   194→        speedY: Math.random() * 0.8 + 0.3,
   195→        speedX: Math.random() * 0.4 - 0.2,
   196→        rotation: Math.random() * Math.PI * 2,
   197→        rotSpeed: (Math.random() - 0.5) * 0.02,
   198→        swayAmp: Math.random() * 0.8 + 0.3,
   199→        swaySpeed: Math.random() * 0.015 + 0.005,
   200→        swayOffset: Math.random() * Math.PI * 2,
   201→        alpha: Math.random() * 0.25 + 0.05,
   202→        type: Math.floor(Math.random() * 3),
   203→        color:
   204→          Math.random() < 0.5
   205→            ? "37,99,235"
   206→            : Math.random() < 0.8
   207→              ? "96,165,250"
   208→              : "34,211,238",
   209→      };
   210→    }
   211→
   212→    const leaves: Particle[] = [];
   213→    for (let i = 0; i < count; i++) leaves.push(createParticle(true));
   214→    particlesRef.current = leaves;
   215→
   216→    function drawParticle(p: Particle, t: number) {
   217→      if (!ctx) return;
   218→      ctx.save();
   219→      ctx.translate(p.x, p.y);
   220→      ctx.rotate(p.rotation);
   221→      ctx.globalAlpha = p.alpha;
   222→
   223→      if (p.type === 0) {
   224→        ctx.beginPath();
   225→        ctx.moveTo(0, -p.size);
   226→        ctx.bezierCurveTo(p.size * 0.8, -p.size * 0.5, p.size * 0.8, p.size * 0.5, 0, p.size);
   227→        ctx.bezierCurveTo(-p.size * 0.8, p.size * 0.5, -p.size * 0.8, -p.size * 0.5, 0, -p.size);
   228→        ctx.fillStyle = `rgba(${p.color},1)`;
   229→        ctx.fill();
   230→      } else if (p.type === 1) {
   231→        ctx.beginPath();
   232→        ctx.arc(0, 0, p.size * 0.5, 0, Math.PI * 2);
   233→        ctx.fillStyle = `rgba(${p.color},1)`;
   234→        ctx.fill();
   235→      } else {
   236→        const s = p.size * 0.6;
   237→        ctx.strokeStyle = `rgba(${p.color},1)`;
   238→        ctx.lineWidth = 0.8;
   239→        ctx.beginPath(); ctx.moveTo(-s, 0); ctx.lineTo(s, 0); ctx.stroke();
   240→        ctx.beginPath(); ctx.moveTo(0, -s); ctx.lineTo(0, s); ctx.stroke();
   241→        ctx.beginPath(); ctx.moveTo(-s * 0.6, -s * 0.6); ctx.lineTo(s * 0.6, s * 0.6); ctx.stroke();
   242→        ctx.beginPath(); ctx.moveTo(s * 0.6, -s * 0.6); ctx.lineTo(-s * 0.6, s * 0.6); ctx.stroke();
   243→      }
   244→      ctx.globalAlpha = 1;
   245→      ctx.restore();
   246→    }
   247→
   248→    function loop() {
   249→      if (!ctx || !canvas) return;
   250→      ctx.clearRect(0, 0, w, h);
   251→      timeRef.current++;
   252→      const t = timeRef.current;
   253→      for (const p of particlesRef.current) {
   254→        p.y += p.speedY;
   255→        p.x += p.speedX + Math.sin(t * p.swaySpeed + p.swayOffset) * p.swayAmp * 0.3;
   256→        p.rotation += p.rotSpeed;
   257→        if (p.y > h + 20) Object.assign(p, createParticle(false));
   258→        if (p.x < -20) p.x = w + 20;
   259→        if (p.x > w + 20) p.x = -20;
   260→        drawParticle(p, t);
   261→      }
   262→      animRef.current = requestAnimationFrame(loop);
   263→    }
   264→    loop();
   265→
   266→    return () => {
   267→      window.removeEventListener("resize", resize);
   268→      cancelAnimationFrame(animRef.current);
   269→    };
   270→  }, []);
   271→
   272→  return <canvas ref={canvasRef} className="fixed inset-0 z-[1] pointer-events-none" aria-hidden="true" />;
   273→}
