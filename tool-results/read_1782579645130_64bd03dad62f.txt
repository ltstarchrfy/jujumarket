   146→interface Particle {
   147→  x: number;
   148→  y: number;
   149→  size: number;
   150→  speedY: number;
   151→  speedX: number;
   152→  rotation: number;
   153→  rotSpeed: number;
   154→  swayAmp: number;
   155→  swaySpeed: number;
   156→  swayOffset: number;
   157→  alpha: number;
   158→  type: number;
   159→  color: string;

   162→// ─── Particle Canvas ─────────────────────────────────────────────────────────
   163→
   164→function ParticleCanvas() {
   165→  const canvasRef = useRef<HTMLCanvasElement>(null);
   166→  const particlesRef = useRef<Particle[]>([]);
   167→  const timeRef = useRef(0);
   168→  const animRef = useRef<number>(0);
   169→
   170→  useEffect(() => {
   171→    const canvas = canvasRef.current;
   172→    if (!canvas) return;
   173→    const ctx = canvas.getContext("2d");
   174→    if (!ctx) return;
   175→
   176→    let w = 0, h = 0;
   177→
   178→    function resize() {
   179→      if (!canvas) return;
   180→      w = canvas.width = window.innerWidth;
   181→      h = canvas.height = window.innerHeight;
   182→    }
   183→    resize();
   184→    window.addEventListener("resize", resize);
   185→
   186→    const isMobile = window.innerWidth < 768;
   187→    const count = isMobile ? 25 : 55;
   188→
   189→    function createParticle(init: boolean): Particle {
   190→      return {
   191→        x: Math.random() * w,
   192→        y: init ? Math.random() * h : -20,
   193→        size: Math.random() * 5 + 2,
   194→        speedY: Math.random() * 0.8 + 0.3,
   195→        speedX: Math.random() * 0.4 - 0.2,
   196→        rotation: Math.random() * Math.PI * 2,
   197→        rotSpeed: (Math.random() - 0.5) * 0.02,
   198→        swayAmp: Math.random() * 0.8 + 0.3,
   199→        swaySpeed: Math.random() * 0.015 + 0.005,
   200→        swayOffset: Math.random() * Math.PI * 2,
   201→        alpha: Math.random() * 0.25 + 0.05,
   202→        type: Math.floor(Math.random() * 3),
   203→        color:
   204→          Math.random() < 0.5
   205→            ? "37,99,235"
   206→            : Math.random() < 0.8
   207→              ? "96,165,250"
   208→              : "34,211,238",
   209→      };
   210→    }
   211→
   212→    const leaves: Particle[] = [];
   213→    for (let i = 0; i < count; i++) leaves.push(createParticle(true));
   214→    particlesRef.current = leaves;
   215→
   216→    function drawParticle(p: Particle, t: number) {
   217→      if (!ctx) return;
   218→      ctx.save();
   219→      ctx.translate(p.x, p.y);
   220→      ctx.rotate(p.rotation);
   221→      ctx.globalAlpha = p.alpha;
   222→
   223→      if (p.type === 0) {
   224→        ctx.beginPath();
   225→        ctx.moveTo(0, -p.size);
   226→        ctx.bezierCurveTo(p.size * 0.8, -p.size * 0.5, p.size * 0.8, p.size * 0.5, 0, p.size);
   227→        ctx.bezierCurveTo(-p.size * 0.8, p.size * 0.5, -p.size * 0.8, -p.size * 0.5, 0, -p.size);
   228→        ctx.fillStyle = `rgba(${p.color},1)`;
   229→        ctx.fill();
   230→      } else if (p.type === 1) {
   231→        ctx.beginPath();
   232→        ctx.arc(0, 0, p.size * 0.5, 0, Math.PI * 2);
   233→        ctx.fillStyle = `rgba(${p.color},1)`;
   234→        ctx.fill();
   235→      } else {
   236→        const s = p.size * 0.6;
   237→        ctx.strokeStyle = `rgba(${p.color},1)`;
   238→        ctx.lineWidth = 0.8;
   239→        ctx.beginPath(); ctx.moveTo(-s, 0); ctx.lineTo(s, 0); ctx.stroke();
   240→        ctx.beginPath(); ctx.moveTo(0, -s); ctx.lineTo(0, s); ctx.stroke();
   241→        ctx.beginPath(); ctx.moveTo(-s * 0.6, -s * 0.6); ctx.lineTo(s * 0.6, s * 0.6); ctx.stroke();
   242→        ctx.beginPath(); ctx.moveTo(s * 0.6, -s * 0.6); ctx.lineTo(-s * 0.6, s * 0.6); ctx.stroke();
   243→      }
   244→      ctx.globalAlpha = 1;
   245→      ctx.restore();
   246→    }
   247→
   248→    function loop() {
   249→      if (!ctx || !canvas) return;
   250→      ctx.clearRect(0, 0, w, h);
   251→      timeRef.current++;
   252→      const t = timeRef.current;
   253→      for (const p of particlesRef.current) {
   254→        p.y += p.speedY;
   255→        p.x += p.speedX + Math.sin(t * p.swaySpeed + p.swayOffset) * p.swayAmp * 0.3;
   256→        p.rotation += p.rotSpeed;
   257→        if (p.y > h + 20) Object.assign(p, createParticle(false));
   258→        if (p.x < -20) p.x = w + 20;
   259→        if (p.x > w + 20) p.x = -20;
   260→        drawParticle(p, t);
   261→      }
   262→      animRef.current = requestAnimationFrame(loop);
   263→    }
   264→    loop();
   265→
   266→    return () => {
   267→      window.removeEventListener("resize", resize);
   268→      cancelAnimationFrame(animRef.current);
   269→    };
   270→  }, []);
   271→
   272→  return <canvas ref={canvasRef} className="fixed inset-0 z-[1] pointer-events-none" aria-hidden="true" />;
   273→}

   588→// ─── Swipeable Feature Carousel (Box + Sound on swipe, Smooth) ──────────────

   590→function FeatureCarousel({ items, theme = "blue" }: {
   591→  items: { num: React.ReactNode; title: string; desc: string; features?: string[] }[];
   592→  theme?: string;
   593→}) {
   594→  const scrollRef = useRef<HTMLDivElement | null>(null);
   595→  const audioRef = useRef<HTMLAudioElement | null>(null);
   596→  const lastIdxRef = useRef<number>(0);
   597→  const rafRef = useRef<number | null>(null);
   598→  const [activeIdx, setActiveIdx] = useState<number>(0);
   599→
   600→  // Color theme
   601→  const themeStyles: Record<string, {
   602→    boxGradient: string;
   603→    boxShadow: string;
   604→    chipGradient: string;
   605→    chipShadow: string;
   606→    accent: string;
   607→    indicatorActive: string;
   608→    labelColor: string;
   609→  }> = {
   610→    blue: {
   611→      boxGradient: "linear-gradient(135deg, rgba(37,99,235,0.10) 0%, rgba(37,99,235,0.02) 60%, var(--ast-bg2) 100%)",
   612→      boxShadow: "0 4px 24px rgba(37,99,235,0.12), 0 1px 3px rgba(0,0,0,0.3)",
   613→      chipGradient: "linear-gradient(135deg, #1e3a5f, #1d4ed8, #60a5fa)",
   614→      chipShadow: "0 2px 10px rgba(37,99,235,0.30)",
   615→      accent: "var(--ast-blue-l)",
   616→      indicatorActive: "var(--ast-blue-l)",
   617→      labelColor: "var(--ast-blue-l)",
   618→    },
   619→    amber: {
   620→      boxGradient: "linear-gradient(135deg, rgba(245,158,11,0.12) 0%, rgba(245,158,11,0.02) 60%, var(--ast-bg2) 100%)",
   621→      boxShadow: "0 4px 24px rgba(245,158,11,0.14), 0 1px 3px rgba(0,0,0,0.3)",
   622→      chipGradient: "linear-gradient(135deg, #92400e, #f59e0b, #fbbf24)",
   623→      chipShadow: "0 2px 10px rgba(245,158,11,0.32)",
   624→      accent: "var(--ast-amber)",
   625→      indicatorActive: "var(--ast-amber)",
   626→      labelColor: "var(--ast-amber)",
   627→    },
   628→  };
   629→  const ts = themeStyles[theme] || themeStyles.blue;
   630→
   631→  // Play swipe sound
   632→  const playSwipeSound = useCallback(() => {
   633→    try {
   634→      if (!audioRef.current) {
   635→        const a = new Audio("/flip.mp3");
   636→        a.volume = 0.3;
   637→        a.preload = "auto";
   638→        audioRef.current = a;
   639→      }
   640→      audioRef.current.currentTime = 0;
   641→      audioRef.current.play().catch(() => {});
   642→    } catch {}
   643→  }, []);
   644→
   645→  // Compute current visible index based on scroll position
   646→  const computeIdx = useCallback(() => {
   647→    const el = scrollRef.current;
   648→    if (!el) return 0;
   649→    const firstChild = el.firstElementChild as HTMLElement | null;
   650→    if (!firstChild) return 0;
   651→    const step = firstChild.offsetWidth + 12;
   652→    const idx = Math.round(el.scrollLeft / step);
   653→    return Math.max(0, Math.min(items.length - 1, idx));
   654→  }, [items.length]);
   655→
   656→  // Throttled scroll handler using rAF
   657→  const handleScroll = useCallback(() => {
   658→    if (rafRef.current != null) return;
   659→    rafRef.current = requestAnimationFrame(() => {
   660→      rafRef.current = null;
   661→      const idx = computeIdx();
   662→      if (idx !== lastIdxRef.current) {
   663→        lastIdxRef.current = idx;
   664→        setActiveIdx(idx);

   668→  }, [computeIdx, playSwipeSound]);

   670→  // Smooth scroll to a specific box index

   672→    const el = scrollRef.current;

   677→    const target = Math.max(0, Math.min(items.length - 1, idx));
   678→    const targetLeft = target * step;
   679→    el.scrollTo({ left: targetLeft, behavior: "smooth" });
   680→    if (lastIdxRef.current !== target) {
   681→      lastIdxRef.current = target;
   682→      setActiveIdx(target);
   683→      playSwipeSound();
   684→    }
   685→  }, [items.length, playSwipeSound]);
   686→
   687→  const scrollByBox = useCallback((dir: 1 | -1) => {
   688→    scrollToIdx(lastIdxRef.current + dir);
   689→  }, [scrollToIdx]);
   690→
   691→  // Cleanup rAF on unmount
   692→  useEffect(() => {
   693→    return () => {
   694→      if (rafRef.current != null) {
   695→        cancelAnimationFrame(rafRef.current);
   696→      }
   697→    };
   698→  }, []);

   721→        style={{
   722→          background: "var(--ast-bg2)",
   723→          borderColor: "var(--ast-border)",
   724→          color: "var(--ast-gray)",

   730→        </svg>

   745→        } as React.CSSProperties}
   746→      >
