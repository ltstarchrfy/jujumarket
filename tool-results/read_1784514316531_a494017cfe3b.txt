   151→  x: number;
   152→  y: number;
   153→  size: number;

   166→// ─── Particle Canvas ─────────────────────────────────────────────────────────
   167→
   168→function ParticleCanvas() {
   169→  const canvasRef = useRef<HTMLCanvasElement>(null);
   170→  const particlesRef = useRef<Particle[]>([]);
   171→  const timeRef = useRef(0);
   172→  const animRef = useRef<number>(0);
   173→
   174→  useEffect(() => {
   175→    const canvas = canvasRef.current;
   176→    if (!canvas) return;
   177→    const ctx = canvas.getContext("2d");
   178→    if (!ctx) return;
   179→
   180→    let w = 0, h = 0;
   181→
   182→    function resize() {
   183→      if (!canvas) return;
   184→      w = canvas.width = window.innerWidth;
   185→      h = canvas.height = window.innerHeight;
   186→    }
   187→    resize();
   188→    window.addEventListener("resize", resize);
   189→
   190→    const isMobile = window.innerWidth < 768;
   191→    const count = isMobile ? 25 : 55;
   192→
   193→    function createParticle(init: boolean): Particle {
   194→      return {
   195→        x: Math.random() * w,
   196→        y: init ? Math.random() * h : -20,
   197→        size: Math.random() * 5 + 2,
   198→        speedY: Math.random() * 0.8 + 0.3,
   199→        speedX: Math.random() * 0.4 - 0.2,
   200→        rotation: Math.random() * Math.PI * 2,
   201→        rotSpeed: (Math.random() - 0.5) * 0.02,
   202→        swayAmp: Math.random() * 0.8 + 0.3,
   203→        swaySpeed: Math.random() * 0.015 + 0.005,
   204→        swayOffset: Math.random() * Math.PI * 2,
   205→        alpha: Math.random() * 0.25 + 0.05,
   206→        type: Math.floor(Math.random() * 3),
   207→        color:
   208→          Math.random() < 0.5
   209→            ? "37,99,235"
   210→            : Math.random() < 0.8
   211→              ? "96,165,250"
   212→              : "34,211,238",
   213→      };
   214→    }
   215→
   216→    const leaves: Particle[] = [];
   217→    for (let i = 0; i < count; i++) leaves.push(createParticle(true));
   218→    particlesRef.current = leaves;
   219→
   220→    function drawParticle(p: Particle, t: number) {
   221→      if (!ctx) return;
   222→      ctx.save();
   223→      ctx.translate(p.x, p.y);
   224→      ctx.rotate(p.rotation);
   225→      ctx.globalAlpha = p.alpha;
   226→
   227→      if (p.type === 0) {
   228→        ctx.beginPath();
   229→        ctx.moveTo(0, -p.size);
   230→        ctx.bezierCurveTo(p.size * 0.8, -p.size * 0.5, p.size * 0.8, p.size * 0.5, 0, p.size);
   231→        ctx.bezierCurveTo(-p.size * 0.8, p.size * 0.5, -p.size * 0.8, -p.size * 0.5, 0, -p.size);
   232→        ctx.fillStyle = `rgba(${p.color},1)`;
   233→        ctx.fill();
   234→      } else if (p.type === 1) {
   235→        ctx.beginPath();
   236→        ctx.arc(0, 0, p.size * 0.5, 0, Math.PI * 2);
   237→        ctx.fillStyle = `rgba(${p.color},1)`;
   238→        ctx.fill();
   239→      } else {
   240→        const s = p.size * 0.6;
   241→        ctx.strokeStyle = `rgba(${p.color},1)`;
   242→        ctx.lineWidth = 0.8;
   243→        ctx.beginPath(); ctx.moveTo(-s, 0); ctx.lineTo(s, 0); ctx.stroke();
   244→        ctx.beginPath(); ctx.moveTo(0, -s); ctx.lineTo(0, s); ctx.stroke();
   245→        ctx.beginPath(); ctx.moveTo(-s * 0.6, -s * 0.6); ctx.lineTo(s * 0.6, s * 0.6); ctx.stroke();
   246→        ctx.beginPath(); ctx.moveTo(s * 0.6, -s * 0.6); ctx.lineTo(-s * 0.6, s * 0.6); ctx.stroke();
   247→      }
   248→      ctx.globalAlpha = 1;
   249→      ctx.restore();
   250→    }
   251→
   252→    function loop() {
   253→      if (!ctx || !canvas) return;
   254→      ctx.clearRect(0, 0, w, h);
   255→      timeRef.current++;
   256→      const t = timeRef.current;
   257→      for (const p of particlesRef.current) {
   258→        p.y += p.speedY;
   259→        p.x += p.speedX + Math.sin(t * p.swaySpeed + p.swayOffset) * p.swayAmp * 0.3;
   260→        p.rotation += p.rotSpeed;
   261→        if (p.y > h + 20) Object.assign(p, createParticle(false));
   262→        if (p.x < -20) p.x = w + 20;
   263→        if (p.x > w + 20) p.x = -20;
   264→        drawParticle(p, t);
   265→      }
   266→      animRef.current = requestAnimationFrame(loop);
   267→    }
   268→    loop();
   269→
   270→    return () => {
   271→      window.removeEventListener("resize", resize);
   272→      cancelAnimationFrame(animRef.current);
   273→    };
   274→  }, []);
   275→
   276→  return <canvas ref={canvasRef} className="fixed inset-0 z-[1] pointer-events-none" aria-hidden="true" />;
   277→}
