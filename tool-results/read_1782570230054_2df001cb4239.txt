    80→// ─── Particle Canvas ─────────────────────────────────────────────────────────
    81→
    82→function ParticleCanvas() {
    83→  const canvasRef = useRef<HTMLCanvasElement>(null);
    84→  const particlesRef = useRef<Particle[]>([]);
    85→  const timeRef = useRef(0);
    86→  const animRef = useRef<number>(0);
    87→
    88→  useEffect(() => {
    89→    const canvas = canvasRef.current;
    90→    if (!canvas) return;
    91→    const ctx = canvas.getContext("2d");
    92→    if (!ctx) return;
    93→
    94→    let w = 0, h = 0;
    95→
    96→    function resize() {
    97→      if (!canvas) return;
    98→      w = canvas.width = window.innerWidth;
    99→      h = canvas.height = window.innerHeight;
   100→    }
   101→    resize();
   102→    window.addEventListener("resize", resize);
   103→
   104→    const isMobile = window.innerWidth < 768;
   105→    const count = isMobile ? 25 : 55;
   106→
   107→    function createParticle(init: boolean): Particle {
   108→      return {
   109→        x: Math.random() * w,
   110→        y: init ? Math.random() * h : -20,
   111→        size: Math.random() * 5 + 2,
   112→        speedY: Math.random() * 0.8 + 0.3,
   113→        speedX: Math.random() * 0.4 - 0.2,
   114→        rotation: Math.random() * Math.PI * 2,
   115→        rotSpeed: (Math.random() - 0.5) * 0.02,
   116→        swayAmp: Math.random() * 0.8 + 0.3,
   117→        swaySpeed: Math.random() * 0.015 + 0.005,
   118→        swayOffset: Math.random() * Math.PI * 2,
   119→        alpha: Math.random() * 0.25 + 0.05,
   120→        type: Math.floor(Math.random() * 3),
   121→        color:
   122→          Math.random() < 0.5
   123→            ? "37,99,235"
   124→            : Math.random() < 0.8
   125→              ? "96,165,250"
   126→              : "34,211,238",
   127→      };
   128→    }
   129→
   130→    const leaves: Particle[] = [];
   131→    for (let i = 0; i < count; i++) leaves.push(createParticle(true));
   132→    particlesRef.current = leaves;
   133→
   134→    function drawParticle(p: Particle, t: number) {
   135→      if (!ctx) return;
   136→      ctx.save();
   137→      ctx.translate(p.x, p.y);
   138→      ctx.rotate(p.rotation);
   139→      ctx.globalAlpha = p.alpha;
   140→
   141→      if (p.type === 0) {
   142→        ctx.beginPath();
   143→        ctx.moveTo(0, -p.size);
   144→        ctx.bezierCurveTo(p.size * 0.8, -p.size * 0.5, p.size * 0.8, p.size * 0.5, 0, p.size);
   145→        ctx.bezierCurveTo(-p.size * 0.8, p.size * 0.5, -p.size * 0.8, -p.size * 0.5, 0, -p.size);
   146→        ctx.fillStyle = `rgba(${p.color},1)`;
   147→        ctx.fill();
   148→      } else if (p.type === 1) {
   149→        ctx.beginPath();
   150→        ctx.arc(0, 0, p.size * 0.5, 0, Math.PI * 2);
   151→        ctx.fillStyle = `rgba(${p.color},1)`;
   152→        ctx.fill();
   153→      } else {
   154→        const s = p.size * 0.6;
   155→        ctx.strokeStyle = `rgba(${p.color},1)`;
   156→        ctx.lineWidth = 0.8;
   157→        ctx.beginPath(); ctx.moveTo(-s, 0); ctx.lineTo(s, 0); ctx.stroke();
   158→        ctx.beginPath(); ctx.moveTo(0, -s); ctx.lineTo(0, s); ctx.stroke();
   159→        ctx.beginPath(); ctx.moveTo(-s * 0.6, -s * 0.6); ctx.lineTo(s * 0.6, s * 0.6); ctx.stroke();
   160→        ctx.beginPath(); ctx.moveTo(s * 0.6, -s * 0.6); ctx.lineTo(-s * 0.6, s * 0.6); ctx.stroke();
   161→      }
   162→      ctx.globalAlpha = 1;
   163→      ctx.restore();
   164→    }
   165→
   166→    function loop() {
   167→      if (!ctx || !canvas) return;
   168→      ctx.clearRect(0, 0, w, h);
   169→      timeRef.current++;
   170→      const t = timeRef.current;
   171→      for (const p of particlesRef.current) {
   172→        p.y += p.speedY;
   173→        p.x += p.speedX + Math.sin(t * p.swaySpeed + p.swayOffset) * p.swayAmp * 0.3;
   174→        p.rotation += p.rotSpeed;
   175→        if (p.y > h + 20) Object.assign(p, createParticle(false));
   176→        if (p.x < -20) p.x = w + 20;
   177→        if (p.x > w + 20) p.x = -20;
   178→        drawParticle(p, t);
   179→      }
   180→      animRef.current = requestAnimationFrame(loop);
   181→    }
   182→    loop();
   183→
   184→    return () => {
   185→      window.removeEventListener("resize", resize);
   186→      cancelAnimationFrame(animRef.current);
   187→    };
   188→  }, []);
   189→
   190→  return <canvas ref={canvasRef} className="fixed inset-0 z-[1] pointer-events-none" aria-hidden="true" />;
   191→}

   218→function StatusBar({ dotColor, label, value, valueColor, valueGradient = false, live = false, bigDot = false, bump = false }: {
   219→  dotColor: string; label: string; value: string; valueColor: string; valueGradient?: boolean; live?: boolean; bigDot?: boolean; bump?: boolean;
   220→}) {
   221→  // WhatsApp-style: uniform gray bar, rounded pill shape, subtle dot (no glow)
   222→  const dotColors: Record<string, string> = {
   223→    green: "var(--ast-green)",
   224→    blue: "var(--ast-blue-l)",
   225→    cyan: "var(--ast-cyan)",
   226→    amber: "var(--ast-amber)",
   227→  };
   228→  const valColors: Record<string, string> = {
   229→    green: "var(--ast-green)",
   230→    blue: "var(--ast-blue-l)",
   231→    cyan: "var(--ast-cyan)",
   232→    white: "#ffffff",
   233→    amber: "var(--ast-amber)",
   234→  };
   235→  const isWhite = valueColor === "white";
   236→  const dotColorVal = dotColors[dotColor] || dotColors.green;
   237→  const dotSize = bigDot ? 10 : 8;
   238→
   239→  return (
   240→    <div
   241→      className="mx-auto flex items-center justify-center gap-2.5 py-2.5 px-5 rounded-full border"
   242→      style={{
   243→        width: "76%",

   246→        borderColor: "rgba(255,255,255,0.04)",
   247→        transition: "all 0.4s cubic-bezier(0.16,1,0.3,1)",
   248→        backdropFilter: "blur(8px)",
   249→      }}
   250→    >
   251→      <span
   252→        className={`rounded-full shrink-0 ${live ? "animate-[dot-ping_1.5s_ease-out_infinite]" : ""}`}
   253→        style={{
   254→          width: `${dotSize}px`,
   255→          height: `${dotSize}px`,
   256→          background: dotColorVal,
   257→          color: dotColorVal,
   258→        }}
   259→      />
   260→      <span className="text-[10.5px] font-semibold tracking-wider" style={{ color: "var(--ast-gray)" }}>{label}</span>
   261→      <span className="text-[10px]" style={{ color: "var(--ast-gray2)" }}>·</span>

   264→        style={{
   265→          color: valColors[valueColor],
   266→          textShadow: "none",
   267→        }}
   268→      >
   269→        {value}
   270→      </span>
   271→    </div>
   272→  );
   273→}

   848→  // ─── Real-time download counter (Firebase + SSE fallback) ──────────────
   849→  // Tries Firebase Realtime Database first; if unavailable, falls back to
   850→  // our own /api/download?stream=1 SSE endpoint. Either way, count updates
   851→  // live whenever anyone (this user or anyone else) clicks a download link.
   852→  const sseRef = useRef<EventSource | null>(null);
   853→  const fbUnsubRef = useRef<(() => void) | null>(null);
   854→
   855→  useEffect(() => {
   856→    let mounted = true;

   858→    async function initFirebase() {

   862→        if (!mounted || !db || !firebaseEnabled) return false;
   863→        const { ref, onValue } = await import("firebase/database");
   864→        const countRef = ref(db, "downloads/total");
   865→        fbUnsubRef.current = onValue(countRef, (snap) => {
   866→          if (!mounted) return;
   867→          const val = typeof snap.val() === "number" ? snap.val() : 1200;
   868→          setDownloadCount((prev) => {
   869→            if (val > prev) setCountBump(true);
   870→            return val;
   871→          });
   872→        });
   873→        return true;
   874→      } catch {
   875→        return false;
   876→      }
   877→    }
   878→
   879→    function initSSE() {
   880→      if (!mounted) return;
   881→      // Don't create duplicate polling loops
   882→      if (sseRef.current) return;
   883→      // Use polling instead of SSE because the preview proxy buffers
   884→      // streaming responses. Polling works through any HTTP proxy.
   885→      let stopped = false;
   886→      // @ts-ignore - we store the stop function in sseRef
   887→      sseRef.current = { close: () => { stopped = true; } } as any;
   888→
   889→      async function poll() {
   890→        // Initial fetch immediately
   891→        try {
   892→          const r = await fetch("/api/download", { cache: "no-store" });
   893→          if (r.ok) {
   894→            const j = await r.json();
   895→            if (j && typeof j.count === "number") {
   896→              setDownloadCount((prev) => {
   897→                if (j.count > prev) setCountBump(true);
   898→                return j.count;
   899→              });
   900→            }
   901→          }
   902→        } catch {}
   903→
   904→        // Then poll every 3 seconds
   905→        while (!stopped && mounted) {
   906→          await new Promise((res) => setTimeout(res, 3000));
   907→          if (stopped || !mounted) break;
   908→          try {
   909→            const r = await fetch("/api/download", { cache: "no-store" });
   910→            if (!r.ok) continue;
   911→            const j = await r.json();
   912→            if (j && typeof j.count === "number") {
   913→              setDownloadCount((prev) => {
   914→                if (j.count !== prev) {
   915→                  if (j.count > prev) setCountBump(true);
   916→                  return j.count;
   917→                }
   918→                return prev;
